# RM object handle (NvHandle).  NV_ESC_RM_ALLOC and NV_ESC_RM_ALLOC_MEMORY
# return it in the hObjectNew field rather than in the ioctl return value, so
# the field is `(inout)`: the executor copies it out after the call (the driver
# overwrites it) and later hClient/hObject fields consume the copied value.
# The generic nvos64/nvos02 fields use `(inout)` so older corpus programs, which
# carry the traced post-call value, still validate (an out resource rejects any
# non-default constant).  The typed root-client/device slices keep `(out)`:
# they are the cycle-free creators that let syz-manager's transitive enable
# analysis accept nvidia_handle at all -- an inout-only resource is treated as
# an input of every alloc too, so nothing can create it.  A single kind is
# intentional: the driver enforces the client/device/memory class, and one
# kind lets an allocation of any class feed any later handle field.  Handles
# are per-client scalars (0xc1d... / 0x5c... / 0xcaf... in the traces), never
# descriptors, so this is a resource purely for ordering/dataflow.
resource nvidia_handle[int32]

syz_open_dev$nvidiactl(dev ptr[in, string["/dev/nvidiactl"]], id intptr, flags flags[open_flags]) fd_nvidiactl
# Opening /dev/nvidia0 runs the GPU init path and costs ~1.7 s in the guest
# (measured 2026-11-28), far above the default 50 ms syscall timeout.  Without
# the per-call timeout the threaded executor schedules the next call while the
# open is still initializing, so an RM allocation issued on nvidiactl blocks on
# the device-init lock and its consumer reads the resource default 0
# (NV_ERR_INVALID_OBJECT_HANDLE).  3 s keeps the call synchronous; the program
# timeout is raised because a split segment can replay up to four such opens.
syz_open_dev$nvidia(dev ptr[in, string["/dev/nvidia#"]], id intptr, flags flags[open_flags]) fd_nvidia_dev (timeout[3000], prog_timeout[10000])
syz_open_dev$nvidia_uvm(dev ptr[in, string["/dev/nvidia-uvm"]], id intptr, flags flags[open_flags]) fd_nvidia_uvm

# Every UVM ioctl number is a bare integer (uvm_ioctl.h: UVM_IOCTL_BASE(i) = i),
# so unlike the _IOWR('F', nr, struct) escape path nothing in the request encodes
# the argument size.  The driver still copies exactly sizeof(struct) to and from
# user space (UVM_ROUTE_CMD_STACK, uvm_api.h), and a short user buffer is read
# past its end, so the modelled size must equal the C sizeof.  The [size[N]]
# attributes below turn that invariant into a compile-time error in syz-sysgen
# instead of a silently truncated copy.  Layouts and sizes come from the real
# headers under open-gpu-kernel-modules/kernel-open/nvidia-uvm.
